```prisma
// packages/prisma/schema/your-business.prisma

// 示例：内容管理系统
model Content {
  id          String   @id @default(cuid())
  title       String
  description String?  @db.Text
  content     String   @db.LongText
  slug        String   @unique
  published   Boolean  @default(false)
  
  // 关联用户和工作空间
  author      User     @relation(fields: [authorId], references: [id])
  authorId    String
  project     Project  @relation(fields: [projectId], references: [id])
  projectId   String
  
  // 分类和标签
  category    Category? @relation(fields: [categoryId], references: [id])
  categoryId  String?
  tags        ContentTag[]
  
  // 统计数据
  views       Int      @default(0)
  likes       Int      @default(0)
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  
  @@index([projectId, published, createdAt])
  @@index([slug])
}

model Category {
  id          String    @id @default(cuid())
  name        String
  slug        String    @unique
  description String?
  color       String?
  
  project     Project   @relation(fields: [projectId], references: [id])
  projectId   String
  contents    Content[]
  
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
  
  @@index([projectId])
}

model ContentTag {
  content   Content @relation(fields: [contentId], references: [id])
  contentId String
  tag       Tag     @relation(fields: [tagId], references: [id])
  tagId     String
  
  @@id([contentId, tagId])
}
```

### 第三步：API 层开发

#### **3.1 创建业务 API 路由**
```typescript
// apps/web/app/api/contents/route.ts

import { withWorkspace } from "@/lib/auth";
import { prisma } from "@dub/prisma";
import { NextResponse } from "next/server";
import { z } from "zod";

const createContentSchema = z.object({
  title: z.string().min(1).max(200),
  description: z.string().optional(),
  content: z.string().min(1),
  slug: z.string().min(1).max(100),
  categoryId: z.string().optional(),
  published: z.boolean().default(false),
});

// GET /api/contents - 获取内容列表
export const GET = withWorkspace(async ({ workspace, searchParams }) => {
  const { page = "1", limit = "10", published, categoryId } = searchParams;
  
  const contents = await prisma.content.findMany({
    where: {
      projectId: workspace.id,
      ...(published !== undefined && { published: published === "true" }),
      ...(categoryId && { categoryId }),
    },
    include: {
      author: {
        select: { id: true, name: true, image: true },
      },
      category: {
        select: { id: true, name: true, slug: true, color: true },
      },
      tags: {
        include: {
          tag: {
            select: { id: true, name: true, color: true },
          },
        },
      },
    },
    orderBy: { createdAt: "desc" },
    take: parseInt(limit),
    skip: (parseInt(page) - 1) * parseInt(limit),
  });
  
  const total = await prisma.content.count({
    where: {
      projectId: workspace.id,
      ...(published !== undefined && { published: published === "true" }),
      ...(categoryId && { categoryId }),
    },
  });
  
  return NextResponse.json({
    contents,
    pagination: {
      page: parseInt(page),
      limit: parseInt(limit),
      total,
      pages: Math.ceil(total / parseInt(limit)),
    },
  });
});

// POST /api/contents - 创建内容
export const POST = withWorkspace(async ({ req, workspace, session }) => {
  const body = await req.json();
  const validatedData = createContentSchema.parse(body);
  
  // 检查 slug 是否已存在
  const existingContent = await prisma.content.findFirst({
    where: {
      slug: validatedData.slug,
      projectId: workspace.id,
    },
  });
  
  if (existingContent) {
    return NextResponse.json(
      { error: "Slug already exists" },
      { status: 400 }
    );
  }
  
  const content = await prisma.content.create({
    data: {
      ...validatedData,
      authorId: session.user.id,
      projectId: workspace.id,
    },
    include: {
      author: {
        select: { id: true, name: true, image: true },
      },
      category: {
        select: { id: true, name: true, slug: true, color: true },
      },
    },
  });
  
  return NextResponse.json(content, { status: 201 });
});
```

#### **3.2 单个资源 API**
```typescript
// apps/web/app/api/contents/[contentId]/route.ts

import { withWorkspace } from "@/lib/auth";
import { prisma } from "@dub/prisma";
import { NextResponse } from "next/server";

// GET /api/contents/[contentId] - 获取单个内容
export const GET = withWorkspace(async ({ params, workspace }) => {
  const { contentId } = params;
  
  const content = await prisma.content.findFirst({
    where: {
      id: contentId,
      projectId: workspace.id,
    },
    include: {
      author: {
        select: { id: true, name: true, image: true },
      },
      category: {
        select: { id: true, name: true, slug: true, color: true },
      },
      tags: {
        include: {
          tag: {
            select: { id: true, name: true, color: true },
          },
        },
      },
    },
  });
  
  if (!content) {
    return NextResponse.json(
      { error: "Content not found" },
      { status: 404 }
    );
  }
  
  // 增加浏览量
  await prisma.content.update({
    where: { id: contentId },
    data: { views: { increment: 1 } },
  });
  
  return NextResponse.json(content);
});

// PUT /api/contents/[contentId] - 更新内容
export const PUT = withWorkspace(async ({ req, params, workspace, session }) => {
  const { contentId } = params;
  const body = await req.json();
  
  // 验证权限
  const existingContent = await prisma.content.findFirst({
    where: {
      id: contentId,
      projectId: workspace.id,
    },
  });
  
  if (!existingContent) {
    return NextResponse.json(
      { error: "Content not found" },
      { status: 404 }
    );
  }
  
  // 检查是否是作者或管理员
  const isAuthor = existingContent.authorId === session.user.id;
  const isAdmin = workspace.users.some(
    (u) => u.userId === session.user.id && u.role === "owner"
  );
  
  if (!isAuthor && !isAdmin) {
    return NextResponse.json(
      { error: "Unauthorized" },
      { status: 403 }
    );
  }
  
  const updatedContent = await prisma.content.update({
    where: { id: contentId },
    data: body,
    include: {
      author: {
        select: { id: true, name: true, image: true },
      },
      category: {
        select: { id: true, name: true, slug: true, color: true },
      },
    },
  });
  
  return NextResponse.json(updatedContent);
});

// DELETE /api/contents/[contentId] - 删除内容
export const DELETE = withWorkspace(async ({ params, workspace, session }) => {
  const { contentId } = params;
  
  const existingContent = await prisma.content.findFirst({
    where: {
      id: contentId,
      projectId: workspace.id,
    },
  });
  
  if (!existingContent) {
    return NextResponse.json(
      { error: "Content not found" },
      { status: 404 }
    );
  }
  
  // 检查权限
  const isAuthor = existingContent.authorId === session.user.id;
  const isAdmin = workspace.users.some(
    (u) => u.userId === session.user.id && u.role === "owner"
  );
  
  if (!isAuthor && !isAdmin) {
    return NextResponse.json(
      { error: "Unauthorized" },
      { status: 403 }
    );
  }
  
  await prisma.content.delete({
    where: { id: contentId },
  });
  
  return NextResponse.json({ success: true });
});
```

#### **4.4 创建 SWR Hook**
```typescript
// apps/web/lib/swr/use-contents.ts

import { fetcher } from "@dub/utils";
import useSWR from "swr";

interface UseContentsParams {
  page?: string;
  published?: string;
  categoryId?: string;
}

export function useContents(params: UseContentsParams = {}) {
  const searchParams = new URLSearchParams();
  
  Object.entries(params).forEach(([key, value]) => {
    if (value) searchParams.set(key, value);
  });
  
  const { data, error, mutate } = useSWR(
    `/api/contents?${searchParams.toString()}`,
    fetcher,
    {
      revalidateOnFocus: false,
    }
  );

  return {
    contents: data,
    loading: !error && !data,
    error,
    mutate,
  };
}

export function useContent(contentId: string) {
  const { data, error, mutate } = useSWR(
    contentId ? `/api/contents/${contentId}` : null,
    fetcher
  );

  return {
    content: data,
    loading: !error && !data,
    error,
    mutate,
  };
}
```
